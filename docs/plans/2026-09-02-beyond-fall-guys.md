# Beyond Fall Guys Implementation Plan


**Goal:** Turn Stumblezone from a faithful four-round Fall Guys clone into a show that is readable on a phone, social while you spectate, varied from show to show, and worth coming back to tomorrow.

**Architecture:** Every new rule lives as a pure module in `src/lib/` (no SDK imports, `node:test` covered) and is wired into the scene by a thin file in `src/systems/`, `src/arena/` or `src/ui/`. Two new rounds (Spotlight, Jump Bar) implement the existing `Round` interface and a seeded per-show draw in `src/lib/schedule.ts` picks which three play before the Hex-Drop finale. UI shapes move from `borderRadius` (not rendered on mobile) to tinted PNG textures generated by a script in `tools/`.

**Tech Stack:** Decentraland SDK7 `@dcl/sdk` 7.27.0, react-ecs, Node 22 (`node:test` + tsx), no runtime dependencies.

**Spec:** `docs/specs/2026-09-02-beyond-fall-guys-design.md`

## Global Constraints

- Node ≥ 22 via Volta; `npm run build && npm test` must be green before every commit.
- Files in `src/lib/` import nothing from `@dcl/*`.
- UI: every `Label` has explicit `width` and `height`; no emoji; only `Button`s carry pointer handlers; `screenInset: 'interactable'`; **no `borderRadius` anywhere in `src/ui/`** (invisible on mobile); no `ParticleSystem`, `LightSource`, `InputModifier`, nine-slice textures.
- Scene: x,z ∈ [0,64], y < 81. New round geometry stays inside `ARENA_RADIUS 26` of `ARENA_CENTER (32, 36)` at `ARENA_Y 24`.
- Schedule constants do not change: `SLOT_SECONDS 120`, `INTRO 20 / PLAY 85 / RESULTS 15`, `ROUND_COUNT 4` acts per show, finale is always Hex-Drop.
- Wallet seed phrase never appears in any file. Deploy is the user's action.
- Commit after every task with a conventional message; never commit with failing tests.

```js
// tools/make-ui.mjs — writes the HUD's plate shapes as white RGBA PNGs the UI tints at runtime.
// Run: node tools/make-ui.mjs
import { deflateSync } from 'node:zlib'
import { mkdirSync, writeFileSync } from 'node:fs'

const crc32 = (() => {
  const t = new Uint32Array(256)
  for (let n = 0; n < 256; n++) { let c = n; for (let k = 0; k < 8; k++) c = c & 1 ? 0xedb88320 ^ (c >>> 1) : c >>> 1; t[n] = c >>> 0 }
  return (buf) => { let c = 0xffffffff; for (const b of buf) c = t[(c ^ b) & 0xff] ^ (c >>> 8); return (c ^ 0xffffffff) >>> 0 }
})()

function chunk(type, data) {
  const len = Buffer.alloc(4); len.writeUInt32BE(data.length)
  const td = Buffer.concat([Buffer.from(type, 'latin1'), data])
  const crc = Buffer.alloc(4); crc.writeUInt32BE(crc32(td))
  return Buffer.concat([len, td, crc])
}

function png(w, h, alphaAt) {
  const raw = Buffer.alloc((w * 4 + 1) * h)
  for (let y = 0; y < h; y++) {
    raw[y * (w * 4 + 1)] = 0
    for (let x = 0; x < w; x++) {
      const o = y * (w * 4 + 1) + 1 + x * 4
      raw[o] = raw[o + 1] = raw[o + 2] = 255
      raw[o + 3] = Math.round(255 * alphaAt(x + 0.5, y + 0.5))
    }
  }
  const ihdr = Buffer.alloc(13)
  ihdr.writeUInt32BE(w, 0); ihdr.writeUInt32BE(h, 4); ihdr[8] = 8; ihdr[9] = 6
  return Buffer.concat([Buffer.from([137, 80, 78, 71, 13, 10, 26, 10]), chunk('IHDR', ihdr), chunk('IDAT', deflateSync(raw)), chunk('IEND', Buffer.alloc(0))])
}

/** 1 inside a rounded rect, with a 1.5px anti-aliased edge. */
function roundedRect(w, h, r) {
  return (x, y) => {
    const dx = Math.max(r - x, x - (w - r), 0)
    const dy = Math.max(r - y, y - (h - r), 0)
    const d = Math.hypot(dx, dy) - r
    return Math.min(1, Math.max(0, 0.5 - d / 1.5))
  }
}

mkdirSync('images/ui', { recursive: true })
writeFileSync('images/ui/pill.png', png(512, 128, roundedRect(512, 128, 64)))
writeFileSync('images/ui/card.png', png(512, 256, roundedRect(512, 256, 48)))
writeFileSync('images/ui/dot.png', png(64, 64, roundedRect(64, 64, 32)))
```

```ts
// src/lib/feed.ts — a short, self-expiring list of one-line events for the corner of the HUD.
export const FEED_TTL_MS = 4000
export const FEED_MAX = 3

export class Feed {
  private items: { text: string; at: number }[] = []

  push(text: string, nowMs: number): void {
    const last = this.items[this.items.length - 1]
    if (last && last.text === text && nowMs - last.at < FEED_TTL_MS) return
    this.items.push({ text, at: nowMs })
    if (this.items.length > FEED_MAX * 2) this.items.splice(0, this.items.length - FEED_MAX * 2)
  }

  /** Newest first, at most FEED_MAX, nothing older than the TTL. */
  visible(nowMs: number): string[] {
    return this.items
      .filter((i) => nowMs - i.at < FEED_TTL_MS)
      .slice(-FEED_MAX)
      .reverse()
      .map((i) => i.text)
  }
}
```

```ts
// src/lib/hype.ts — the crowd's temperature. Cheers in a sliding window; enough of them and the stadium erupts.
export const HYPE_WINDOW_MS = 10000
export const HYPE_CHEERS = 5
export const HYPE_COOLDOWN_MS = 20000

export class Hype {
  private cheers: number[] = []
  private lastWildAt = -Infinity

  cheer(nowMs: number): void {
    this.cheers.push(nowMs)
    this.prune(nowMs)
  }

  private prune(nowMs: number): void {
    this.cheers = this.cheers.filter((t) => nowMs - t < HYPE_WINDOW_MS)
  }

  level(nowMs: number): number {
    this.prune(nowMs)
    return Math.min(1, this.cheers.length / HYPE_CHEERS)
  }

  /** True once when the crowd tips over; then quiet for the cooldown. */
  consumeWild(nowMs: number): boolean {
    if (nowMs - this.lastWildAt < HYPE_COOLDOWN_MS) return false
    if (this.level(nowMs) < 1) return false
    this.lastWildAt = nowMs
    this.cheers = []
    return true
  }
}
```

```ts
export function fieldLine(names: string[], max = 3): string {
  if (names.length === 0) return ''
  const shown = names.slice(0, max).join(', ')
  const rest = names.length - max
  return 'IN: ' + shown + (rest > 0 ? ' +' + rest : '')

type Out = { name: string; outMs: number | null }

/** outMs null means survived. The rival is whoever finished nearest to you on the clock. */
export function rivalry(me: Out, others: Out[]): string {
```

```ts
import { DISC_RADIUS } from '../config'
import { mulberry32 } from './prng'

export const SPOT_HIT_SECONDS = 1.2
export const SPOT_WARN_SECONDS = 0.6
export const SPOT_RADIUS = 1.8
export const SPOT_THIRD_AT = 45

export function spotCount(elapsed: number): number { return elapsed >= SPOT_THIRD_AT ? 3 : 2 }
export function spotSpeed(elapsed: number): number { return elapsed >= SPOT_THIRD_AT ? 1.5 : 1 }

/** A Lissajous figure per spot: two incommensurate frequencies and a seeded phase, so it never settles. */
export function spotCentre(seed: number, i: number, t: number): { x: number; z: number } {
  const rng = mulberry32((seed ^ 0x51907) + i * 7919)
  const fx = 0.11 + rng() * 0.07, fz = 0.13 + rng() * 0.07
  const px = rng() * Math.PI * 2, pz = rng() * Math.PI * 2
  const r = DISC_RADIUS - 2
  const s = spotSpeed(t)
  const tt = t < SPOT_THIRD_AT ? t : SPOT_THIRD_AT + (t - SPOT_THIRD_AT) * s
  return { x: r * Math.sin(fx * tt * Math.PI * 2 + px), z: r * Math.sin(fz * tt * Math.PI * 2 + pz) }
}

  private inLight = 0
  update(inside: boolean, dt: number): 'ok' | 'warn' | 'hit' {
    if (!inside) { this.inLight = 0; return 'ok' }
    this.inLight += dt
    if (this.inLight >= SPOT_HIT_SECONDS) { this.inLight = 0; return 'hit' }
    return this.inLight >= SPOT_HIT_SECONDS - SPOT_WARN_SECONDS ? 'warn' : 'ok'
  }
}
```
